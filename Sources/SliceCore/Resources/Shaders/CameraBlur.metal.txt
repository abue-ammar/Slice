#include <CoreImage/CoreImage.h>
using namespace metal;

[[ stitchable ]] float4 cameraMotionBlur(coreimage::sampler layer, float4 rectA, float4 rectB, float count,
                                         coreimage::destination dest) {
    float2 p = dest.coord();
    int n = max(int(count), 1);
    float4 sum = float4(0.0);
    for (int i = 0; i < n; i++) {
        float f = n > 1 ? float(i) / float(n - 1) : 0.0;
        float4 r = mix(rectA, rectB, f);
        float2 q = rectA.xy + (p - r.xy) / r.zw * rectA.zw;
        sum += layer.sample(layer.transform(q));
    }
    return sum / float(n);
}

// radii: bottom-left, bottom-right, top-left, top-right, in Core Image's y-up space.
[[ stitchable ]] float4 cornerRoundedRect(float4 rect, float4 radii, coreimage::destination dest) {
    float2 extent = rect.zw * 0.5;
    float2 p = dest.coord() - (rect.xy + extent);
    float r = p.y > 0.0 ? (p.x > 0.0 ? radii.w : radii.z) : (p.x > 0.0 ? radii.y : radii.x);
    float2 q = abs(p) - extent + r;
    float d = min(max(q.x, q.y), 0.0) + length(max(q, 0.0)) - r;
    return float4(clamp(0.5 - d, 0.0, 1.0));
}
