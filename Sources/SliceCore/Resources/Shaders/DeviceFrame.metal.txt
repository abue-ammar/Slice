#include <metal_stdlib>
using namespace metal;

// Composites a device's live screen into its frame image, bottom to top:
//   1. the live screen, aspect-filled into the opening
//   2. clipped by the opening's mask (measured from the frame image's transparency)
//   3. the frame image (bezel, Dynamic Island, buttons) over it, premultiplied
// Everything outside the device stays transparent.
//
// Kept as .txt and compiled at runtime with MTLDevice.makeLibrary(source:), like the other shaders in this package,
// so it builds the same under SwiftPM and Xcode.

struct DeviceFrameVertexOut {
    float4 position [[position]];
    float2 uv;
};

struct DeviceFrameUniforms {
    float4 destination;
    float4 screenRect;
    float4 streamCrop;
    uint landscape;
    uint hasScreen;
};

vertex DeviceFrameVertexOut deviceFrameVertex(uint vid [[vertex_id]], constant DeviceFrameUniforms &u [[buffer(0)]]) {
    float2 uv = float2(float(vid & 1), float(vid >> 1));
    float2 position = float2(mix(u.destination.x, u.destination.z, uv.x), mix(u.destination.y, u.destination.w, uv.y));
    DeviceFrameVertexOut out;
    out.position = float4(position, 0.0, 1.0);
    out.uv = uv;
    return out;
}

fragment float4 deviceFrameFragment(DeviceFrameVertexOut in [[stage_in]],
                                    constant DeviceFrameUniforms &u [[buffer(0)]],
                                    texture2d<float> frame [[texture(0)]],
                                    texture2d<float> mask [[texture(1)]],
                                    texture2d<float> screen [[texture(2)]],
                                    sampler smooth [[sampler(0)]]) {
    float2 uv = in.uv;
    float2 framePoint = u.landscape != 0 ? float2(1.0 - uv.y, uv.x) : uv;
    float4 bezel = frame.sample(smooth, framePoint);
    float open = mask.sample(smooth, framePoint).r;

    float4 display = float4(0.0, 0.0, 0.0, 1.0);
    if (u.hasScreen != 0) {
        float2 s = clamp((uv - u.screenRect.xy) / u.screenRect.zw, 0.0, 1.0);
        display = float4(screen.sample(smooth, u.streamCrop.xy + s * u.streamCrop.zw).rgb, 1.0);
    }
    return bezel + display * open * (1.0 - bezel.a);
}

// MARK: - Morph

// The gooey morph from one look of a device to another (a different frame or colour), after react-native-morph-view.
// Both looks are drawn, crossfaded and blurred together; a soft alpha threshold then turns the blurred, half-clear edge
// back into a crisp one, so the two outlines fuse and flow into each other like liquid instead of dissolving.

kernel void deviceMorphCombine(texture2d<float, access::read> from [[texture(0)]],
                               texture2d<float, access::read> to [[texture(1)]],
                               texture2d<float, access::write> out [[texture(2)]],
                               constant float &progress [[buffer(0)]],
                               uint2 gid [[thread_position_in_grid]]) {
    if (gid.x >= out.get_width() || gid.y >= out.get_height()) return;
    float4 a = from.read(gid) * (1.0 - progress);
    float4 b = to.read(gid) * progress;
    out.write(b + a * (1.0 - b.a), gid);
}

struct DeviceMorphVertexOut {
    float4 position [[position]];
    float2 uv;
};

vertex DeviceMorphVertexOut deviceMorphVertex(uint vid [[vertex_id]]) {
    float2 uv = float2(float(vid & 1), float(vid >> 1));
    DeviceMorphVertexOut out;
    out.position = float4(uv.x * 2.0 - 1.0, 1.0 - uv.y * 2.0, 0.0, 1.0);
    out.uv = uv;
    return out;
}

fragment float4 deviceMorphThreshold(DeviceMorphVertexOut in [[stage_in]],
                                     texture2d<float> blurred [[texture(0)]],
                                     constant float &strength [[buffer(0)]],
                                     sampler smooth [[sampler(0)]]) {
    float4 c = blurred.sample(smooth, in.uv);
    if (c.a < 0.001) return float4(0.0);
    float a = mix(c.a, smoothstep(0.45, 0.55, c.a), strength);
    return float4(c.rgb / c.a * a, a);
}
