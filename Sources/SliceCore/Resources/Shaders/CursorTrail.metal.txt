#include <CoreImage/CoreImage.h>
using namespace metal;

// The sprites come in sRGB, unconverted: each sample is filtered as stored, then brought into the linear working space,
// as Core Image does for a transformed image.
static float4 linearised(float4 c) {
    if (c.a <= 0.0) return float4(0.0);
    float3 u = c.rgb / c.a;
    u = select(pow((u + 0.055) / 1.055, 2.4), u / 12.92, u <= 0.04045);
    return float4(u * c.a, c.a);
}

// One texel of `poses` per pose: where the hotspot lands, then cos and sin of the pose's angle over its scale.
[[ stitchable ]] float4 cursorTrail(coreimage::sampler current, coreimage::sampler previous, coreimage::sampler poses,
                                    float count, float fade, float2 currentHotspot, float2 previousHotspot,
                                    coreimage::destination dest) {
    float2 p = dest.coord();
    int n = max(int(count), 1);
    float4 sum = float4(0.0);
    for (int i = 0; i < n; i++) {
        float4 pose = poses.sample(poses.transform(float2(float(i) + 0.5, 0.5)));
        float2 d = p - pose.xy;
        float2 local = float2(pose.z * d.x + pose.w * d.y, pose.z * d.y - pose.w * d.x);
        float4 c = linearised(current.sample(current.transform(local + currentHotspot)));
        if (fade < 1.0) {
            c = c * fade + linearised(previous.sample(previous.transform(local + previousHotspot))) * (1.0 - fade);
        }
        sum += c;
    }
    return sum / float(n);
}
